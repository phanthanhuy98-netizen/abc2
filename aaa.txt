function haversine(lat1,lon1,lat2,lon2){
  const R=6371000;
  const toRad=d=>d*Math.PI/180;
  const dLat=toRad(lat2-lat1), dLon=toRad(lon2-lon1);
  const a=Math.sin(dLat/2)**2+Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLon/2)**2;
  return R*2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));
}

function getPosition(){
  return new Promise((resolve,reject)=>{
    if(!navigator.geolocation){ reject(new Error('no-geo')); return; }
    navigator.geolocation.getCurrentPosition(
      pos=>resolve({lat:pos.coords.latitude,lng:pos.coords.longitude}),
      err=>reject(err),
      {enableHighAccuracy:true, timeout:10000}
    );

function todaysCheckins(empId){
  const tk = todayKey();
  return STATE.checkins.filter(c=>c.empId===empId && c.time.slice(0,10)===tk);
}
function todaysAttendance(empId){
  const tk = todayKey();
  return STATE.attendance.filter(a=>a.empId===empId && a.time.slice(0,10)===tk);
}
function empRouteStores(empId){
  const ids = STATE.routes[empId] || [];
  return ids.map(id=>STATE.stores.find(s=>s.id===id)).filter(Boolean);
}

  return `

/* ====================== RENDER: REPORTS ====================== */
function reportsHtml(){
  const range = STATE.reportRange;
  const start = rangeStart(range);
  const checkins = STATE.checkins.filter(c=> !start || new Date(c.time)>=start);
  const orders = STATE.orders.filter(o=> !start || new Date(o.time)>=start);
  const totalRevenue = orders.reduce((s,o)=>s+o.total,0);

  const byEmp = STATE.employees.map(e=>{
    const ec = checkins.filter(c=>c.empId===e.id);
    const eo = orders.filter(o=>o.empId===e.id);
    return { name:e.name, visits:ec.length, orderCount:eo.length, revenue: eo.reduce((s,o)=>s+o.total,0) };
  }).sort((a,b)=>b.revenue-a.revenue);

  const byStore = STATE.stores.map(s=>{
    const sc = checkins.filter(c=>c.storeId===s.id);
    const so = orders.filter(o=>o.storeId===s.id);
    return { name:s.name, category:s.category, visits:sc.length, orderCount:so.length, revenue: so.reduce((s2,o)=>s2+o.total,0) };
  }).sort((a,b)=>b.revenue-a.revenue);

  const rangeLabel = {today:'Hôm nay', week:'Tuần này', month:'Tháng này', all:'Toàn bộ'};

  return `
    <div class="section-head"><div><h2>Báo cáo</h2><div class="sub">Tổng hợp lượt ghé quầy và doanh thu</div></div></div>
    <div class="range-tabs">
      ${['today','week','month','all'].map(r=>`<div class="tab ${range===r?'active':''}" data-action="report-range" data-range="${r}">${rangeLabel[r]}</div>`).join('')}
    </div>
    <div class="kpis">
      <div class="kpi"><div class="v">${checkins.length}</div><div class="l">Lượt ghé quầy</div></div>
      <div class="kpi amber"><div class="v">${orders.length}</div><div class="l">Đơn hàng</div></div>
      <div class="kpi green"><div class="v">${fmtVND(totalRevenue)}</div><div class="l">Doanh thu</div></div>
      <div class="kpi"><div class="v">${STATE.employees.length}</div><div class="l">Nhân viên</div></div>
    </div>

    <div class="section-head"><h2 style="font-size:16px;">Theo nhân viên</h2></div>
    <div style="overflow-x:auto;margin-bottom:24px;">
    <table class="data-table">
      <thead><tr><th>Nhân viên</th><th>Lượt ghé</th><th>Đơn hàng</th><th>Doanh thu</th></tr></thead>
      <tbody>
        ${byEmp.length? byEmp.map(r=>`<tr><td>${esc(r.name)}</td><td>${r.visits}</td><td>${r.orderCount}</td><td>${fmtVND(r.revenue)}</td></tr>`).join('') : `<tr><td colspan="4" style="text-align:center;color:var(--muted);">Chưa có dữ liệu</td></tr>`}
      </tbody>
    </table>
    </div>

    <div class="section-head"><h2 style="font-size:16px;">Theo quầy</h2></div>
    <div style="overflow-x:auto;">
    <table class="data-table">
      <thead><tr><th>Quầy</th><th>Loại</th><th>Lượt ghé</th><th>Đơn hàng</th><th>Doanh thu</th></tr></thead>
      <tbody>
        ${byStore.length? byStore.map(r=>`<tr><td>${esc(r.name)}</td><td><span class="cat-chip ${r.category||'normal'}">${catLabel(r.category)}</span></td><td>${r.visits}</td><td>${r.orderCount}</td><td>${fmtVND(r.revenue)}</td></tr>`).join('') : `<tr><td colspan="5" style="text-align:center;color:var(--muted);">Chưa có dữ liệu</td></tr>`}
      </tbody>
    </table>
    </div>
  `;
}

/* ====================== EVENTS ====================== */
function bindEvents(){
  document.querySelectorAll('[data-action]').forEach(el=>{
    const evt = el.tagName==='SELECT' ? 'change' : (el.tagName==='INPUT' && (el.type==='checkbox'||el.type==='file')) ? 'change' : 'click';
    el.addEventListener(evt, handleAction);
  });

  const selEmp = document.getElementById('sel-emp');
  if(selEmp) selEmp.addEventListener('change', e=>{ 
    STATE.builderEmployee = e.target.value; 
    STATE.builderStores = STATE.routes[e.target.value] ? [...STATE.routes[e.target.value]] : [];
    render(); 
  });

  const nsName = document.getElementById('new-store-name');
  if(nsName) nsName.addEventListener('input', e=>{
    STATE.newStore.name = e.target.value;
    const btn = document.getElementById('store-save-btn');
    if(btn) btn.disabled = !STATE.newStore.name;
  });
  const nsAddr = document.getElementById('new-store-address');
  if(nsAddr) nsAddr.addEventListener('input', e=>{ STATE.newStore.address = e.target.value; });

  const efName = document.getElementById('emp-form-name');
  if(efName) efName.addEventListener('input', e=>{
    STATE.empForm.name = e.target.value;
    const btn = document.querySelector('[data-action="save-employee"]');
    if(btn) btn.disabled = !STATE.empForm.name;
  });
  const efRole = document.getElementById('emp-form-role');
  if(efRole) efRole.addEventListener('input', e=>{ STATE.empForm.role = e.target.value; });
  const efPhone = document.getElementById('emp-form-phone');
  if(efPhone) efPhone.addEventListener('input', e=>{ STATE.empForm.phone = e.target.value; });

  const pfName = document.getElementById('prod-form-name');
  if(pfName) pfName.addEventListener('input', e=>{
    STATE.productForm.name = e.target.value;
    const btn = document.querySelector('[data-action="save-product"]');
    if(btn) btn.disabled = !STATE.productForm.name;
  });
  const pfPrice = document.getElementById('prod-form-price');
  if(pfPrice) pfPrice.addEventListener('input', e=>{ STATE.productForm.price = e.target.value; });

  const detailNote = document.getElementById('detail-note');
  if(detailNote) detailNote.addEventListener('input', e=>{ STATE.detailDraft.note = e.target.value; });

  document.querySelectorAll('.order-qty-input').forEach(inp=>{
    inp.addEventListener('input', e=>{
      STATE.detailDraft.qty[e.target.dataset.pid] = e.target.value;
      const total = STATE.products.reduce((sum,p)=> sum + (Number(STATE.detailDraft.qty[p.id])||0)*p.price, 0);
      const totalEl = document.getElementById('order-total-live');
      if(totalEl) totalEl.textContent = fmtVND(total);
    });
  });
}

async function handleAction(e){
  const el = e.currentTarget;
  const action = el.dataset.action;

  if(action==='pick-role'){
    const role = el.dataset.role;
    if(role==='manager'){ STATE.screen='manager'; STATE.managerTab='dashboard'; render(); return; }
    if(role==='employee'){
      document.getElementById('loginSub').innerHTML = employeePickHtml();
      bindEvents();
      return;
    }
  }
  if(action==='back-role'){ document.getElementById('loginSub').innerHTML=''; return; }
  if(action==='login-employee'){
    const emp = STATE.employees.find(x=>x.id===el.dataset.id);
    STATE.currentEmployee = emp; STATE.screen='employee'; STATE.detailStoreId=null; render(); return;
  }
  if(action==='logout'){ STATE.screen='login'; STATE.currentEmployee=null; render(); return; }
  if(action==='mtab'){ STATE.managerTab = el.dataset.tab; render(); return; }

  if(action==='toggle-shift'){
    const emp = STATE.currentEmployee;
    const att = todaysAttendance(emp.id);
    const last = att[att.length-1];
    const goingOn = !(last && last.type==='in');
    try{
      const pos = await getPosition();
      STATE.attendance.push({empId:emp.id, type: goingOn?'in':'out', time: new Date().toISOString(), lat:pos.lat, lng:pos.lng});
    }catch(err){
      STATE.attendance.push({empId:emp.id, type: goingOn?'in':'out', time: new Date().toISOString(), lat:null, lng:null});
    }
    saveStored(KEYS.attendance, STATE.attendance);
    showToast(goingOn? 'Đã vào ca' : 'Đã tan ca');
    render(); return;
  }

  if(action==='checkin'){
    const storeId = el.dataset.store;
    const store = STATE.stores.find(s=>s.id===storeId);
    const emp = STATE.currentEmployee;
    el.disabled = true; el.textContent = 'Đang lấy vị trí...';
    try{
      const pos = await getPosition();
      const dist = haversine(pos.lat,pos.lng,store.lat,store.lng);
      let proceed = true;
      if(dist>200){
        proceed = confirm(`Bạn đang cách quầy "${store.name}" khoảng ${Math.round(dist)}m (>200m). Vẫn xác nhận ghé quầy?`);
      }
      if(proceed){
        STATE.checkins.push({empId:emp.id, storeId, time:new Date().toISOString(), lat:pos.lat, lng:pos.lng, distance:dist, outcome:'', note:'', photo:null});
        saveStored(KEYS.checkins, STATE.checkins);
        showToast(`Đã ghé "${store.name}" — cách ${Math.round(dist)}m`);
      }
    }catch(err){
      showToast('Không lấy được vị trí GPS. Hãy bật định vị trình duyệt.');
    }
    render(); return;
  }

  if(action==='toggle-detail'){
    const storeId = el.dataset.store;
    if(STATE.detailStoreId===storeId){
      STATE.detailStoreId = null;
      STATE.detailDraft = {outcome:'', note:'', photo:null, qty:{}};
    } else {
      STATE.detailStoreId = storeId;
      const emp = STATE.currentEmployee;
      const c = todaysCheckins(emp.id).find(c=>c.storeId===storeId);
      
      // Load existing order qty if exists
      const existingOrder = STATE.orders.find(o=>o.empId===emp.id && o.storeId===storeId && o.time.slice(0,10)===todayKey());
      const initialQty = {};
      if(existingOrder && existingOrder.items){
        existingOrder.items.forEach(it=>{ initialQty[it.productId] = it.qty; });
      }

      STATE.detailDraft = { outcome:(c&&c.outcome)||'', note:(c&&c.note)||'', photo:(c&&c.photo)||null, qty:initialQty };
    }
    render(); return;
  }
  if(action==='detail-outcome'){ STATE.detailDraft.outcome = el.value; return; }
  if(action==='detail-photo'){
    const file = el.files && el.files[0];
    if(!file) return;
    const reader = new FileReader();
    reader.onload = ()=>{ STATE.detailDraft.photo = reader.result; render(); };
    reader.readAsDataURL(file);
    return;
  }
  if(action==='save-detail'){
    const storeId = el.dataset.store, empId = el.dataset.emp;
    const tk = todayKey();
    const c = STATE.checkins.find(x=>x.empId===empId && x.storeId===storeId && x.time.slice(0,10)===tk);
    if(c){ c.outcome = STATE.detailDraft.outcome; c.note = STATE.detailDraft.note; c.photo = STATE.detailDraft.photo; }
    
    const total = STATE.products.reduce((sum,p)=> sum + (Number(STATE.detailDraft.qty[p.id])||0)*p.price, 0);
    
    // Gỡ bỏ đơn cũ của lượt ghé hôm nay nếu có để tránh trùng lặp
    STATE.orders = STATE.orders.filter(o=> !(o.empId===empId && o.storeId===storeId && o.time.slice(0,10)===tk));

    if(total > 0){
      const items = STATE.products.filter(p=>Number(STATE.detailDraft.qty[p.id])>0)
        .map(p=>({productId:p.id, name:p.name, price:p.price, qty:Number(STATE.detailDraft.qty[p.id])}));
      STATE.orders.push({id:'o'+Math.random().toString(36).slice(2,8), empId, storeId, time:new Date().toISOString(), items, total});
    }

    saveStored(KEYS.checkins, STATE.checkins);
    saveStored(KEYS.orders, STATE.orders);
    STATE.detailStoreId = null;
    STATE.detailDraft = {outcome:'', note:'', photo:null, qty:{}};
    showToast('Đã lưu kết quả ghé quầy');
    render(); return;
  }

  if(action==='toggle-store'){
    const id = el.value;
    if(el.checked){ if(!STATE.builderStores.includes(id)) STATE.builderStores.push(id); }
    else{ STATE.builderStores = STATE.builderStores.filter(x=>x!==id); }
    return;
  }
  if(action==='save-route'){
    if(!STATE.builderEmployee) return;
    STATE.routes[STATE.builderEmployee] = [...STATE.builderStores];
    saveStored(KEYS.routes, STATE.routes);
    showToast('Đã lưu tuyến');
    STATE.builderEmployee=''; STATE.builderStores=[];
    render(); return;
  }
  if(action==='clear-route'){
    const empId = el.dataset.emp;
    delete STATE.routes[empId];
    saveStored(KEYS.routes, STATE.routes);
    render(); return;
  }

  if(action==='edit-employee'){
    const emp = STATE.employees.find(x=>x.id===el.dataset.id);
    if(emp) STATE.empForm = {id:emp.id, name:emp.name, role:emp.role||'', phone:emp.phone||''};
    render(); return;
  }
  if(action==='cancel-employee-edit'){ STATE.empForm = {id:null,name:'',role:'',phone:''}; render(); return; }
  if(action==='save-employee'){
    const f = STATE.empForm;
    if(!f.name) return;
    if(f.id){
      const idx = STATE.employees.findIndex(x=>x.id===f.id);
      if(idx>-1) STATE.employees[idx] = {...STATE.employees[idx], name:f.name, role:f.role, phone:f.phone};
    } else {
      const id = 'nv'+Math.random().toString(36).slice(2,8);
      STATE.employees.push({id, name:f.name, role: f.role || 'Sales ngoài thị trường', phone:f.phone});
    }
    saveStored(KEYS.employees, STATE.employees);
    const wasEdit = !!f.id;
    STATE.empForm = {id:null,name:'',role:'',phone:''};
    showToast(wasEdit? 'Đã lưu thay đổi' : 'Đã thêm nhân viên');
    render(); return;
  }
  if(action==='delete-employee'){
    const id = el.dataset.id;
    const emp = STATE.employees.find(x=>x.id===id);
    if(!emp) return;
    if(!confirm(`Xoá nhân viên "${emp.name}"? Tuyến hiện tại của nhân viên này sẽ bị xoá.`)) return;
    STATE.employees = STATE.employees.filter(x=>x.id!==id);
    delete STATE.routes[id];
    saveStored(KEYS.employees, STATE.employees);
    saveStored(KEYS.routes, STATE.routes);
    showToast('Đã xoá nhân viên');
    render(); return;
  }

  if(action==='edit-product'){
    const p = STATE.products.find(x=>x.id===el.dataset.id);
    if(p) STATE.productForm = {id:p.id, name:p.name, price:String(p.price)};
    render(); return;
  }
  if(action==='cancel-product-edit'){ STATE.productForm = {id:null,name:'',price:''}; render(); return; }
  if(action==='save-product'){
    const f = STATE.productForm;
    if(!f.name) return;
    const price = Number(String(f.price).replace(/[^0-9]/g,'')) || 0;
    if(f.id){
      const idx = STATE.products.findIndex(x=>x.id===f.id);
      if(idx>-1) STATE.products[idx] = {...STATE.products[idx], name:f.name, price};
    } else {
      const id = 'p'+Math.random().toString(36).slice(2,8);
      STATE.products.push({id, name:f.name, price});
    }
    saveStored(KEYS.products, STATE.products);
    const wasEdit = !!f.id;
    STATE.productForm = {id:null,name:'',price:''};
    showToast(wasEdit? 'Đã lưu thay đổi' : 'Đã thêm sản phẩm');
    render(); return;
  }
  if(action==='delete-product'){
    const id = el.dataset.id;
    const p = STATE.products.find(x=>x.id===id);
    if(!p) return;
    if(!confirm(`Xoá sản phẩm "${p.name}"?`)) return;
    STATE.products = STATE.products.filter(x=>x.id!==id);
    saveStored(KEYS.products, STATE.products);
    showToast('Đã xoá sản phẩm');
    render(); return;
  }

  if(action==='set-store-cat'){ STATE.newStore.category = el.value; return; }
  if(action==='edit-store'){
    const s = STATE.stores.find(x=>x.id===el.dataset.id);
    if(s) STATE.newStore = {id:s.id, name:s.name, address:s.address, lat:s.lat, lng:s.lng, category:s.category||'normal'};
    render(); return;
  }
  if(action==='cancel-store-edit'){ STATE.newStore = {id:null,name:'',address:'',lat:null,lng:null,category:'normal'}; render(); return; }
  if(action==='delete-store'){
    const id = el.dataset.id;
    const s = STATE.stores.find(x=>x.id===id);
    if(!s) return;
    if(!confirm(`Xoá quầy "${s.name}"? Quầy sẽ được gỡ khỏi mọi tuyến.`)) return;
    STATE.stores = STATE.stores.filter(x=>x.id!==id);
    Object.keys(STATE.routes).forEach(empId=>{ STATE.routes[empId] = (STATE.routes[empId]||[]).filter(sid=>sid!==id); });
    STATE.builderStores = STATE.builderStores.filter(sid=>sid!==id);
    saveStored(KEYS.stores, STATE.stores);
    saveStored(KEYS.routes, STATE.routes);
    showToast('Đã xoá quầy');
    render(); return;
  }
  if(action==='get-store-gps'){
    try{
      const pos = await getPosition();
      STATE.newStore.lat = pos.lat; STATE.newStore.lng = pos.lng;
      showToast('Đã lấy toạ độ hiện tại');
    }catch(err){ showToast('Không lấy được GPS'); }
    render(); return;
  }
  if(action==='add-store'){
    const f = STATE.newStore;
    if(!f.name) return;
    if(f.id){
      const idx = STATE.stores.findIndex(x=>x.id===f.id);
      if(idx>-1){
        STATE.stores[idx] = {...STATE.stores[idx], name:f.name, address: f.address || '(chưa có địa chỉ)',
          lat: f.lat!=null? f.lat : STATE.stores[idx].lat, lng: f.lng!=null? f.lng : STATE.stores[idx].lng, category:f.category||'normal'};
      }
      showToast('Đã lưu thay đổi');
    } else {
      const id = 's' + Math.random().toString(36).slice(2,8);
      STATE.stores.push({
        id, name:f.name, address: f.address || '(chưa có địa chỉ)',
        lat: f.lat || (11.53 + (Math.random()-0.5)*0.02),
        lng: f.lng || (106.90 + (Math.random()-0.5)*0.02),
        category: f.category || 'normal'
      });
      showToast('Đã thêm quầy mới');
    }
    saveStored(KEYS.stores, STATE.stores);
    STATE.newStore = {id:null, name:'',address:'',lat:null,lng:null,category:'normal'};
    render(); return;
  }

  if(action==='report-range'){ STATE.reportRange = el.dataset.range; render(); return; }

  if(action==='reset-day'){
    if(!confirm('Xoá toàn bộ check-in, đơn hàng và chấm công hôm nay để bắt đầu ngày mới?')) return;
    STATE.checkins = []; STATE.attendance = []; STATE.orders = [];
    saveStored(KEYS.checkins, []); saveStored(KEYS.attendance, []); saveStored(KEYS.orders, []);
    showToast('Đã bắt đầu ngày mới');
    render(); return;
  }
}
